C program to calculate FOLLOW sets

#include <stdio.h>
#include <ctype.h>

char p[6][10] = {
    "E=E+T",
    "E=T",
    "T=T*F",
    "T=F",
    "F=(E)",
    "F=id"
};

void follow(char c)
{
    int i, j;

    if (c == 'E')
        printf("$ ");

    for (i = 0; i < 6; i++)
    {
        for (j = 2; p[i][j] != '\0'; j++)
        {
            if (p[i][j] == c)
            {
                if (p[i][j+1] != '\0')
                {
                    if (!isupper(p[i][j+1]))
                        printf("%c ", p[i][j+1]);
                }
                else if (p[i][0] != c)
                    follow(p[i][0]);
            }
        }
    }
}

int main()
{
    printf("FOLLOW(E) = { ");
    follow('E');
    printf("}\n");

    printf("FOLLOW(T) = { ");
    follow('T');
    printf("}\n");

    printf("FOLLOW(F) = { ");
    follow('F');
    printf("}\n");

    return 0;
}

Simple YACC Program for Infix to Postfix Conversion
1. YACC file (infix.y)

%{
#include <stdio.h>
int yylex();
void yyerror(char *s);
%}

%token ID
%left '+' '-'
%left '*' '/'

%%

E : E '+' E  { printf("+ "); }
  | E '-' E  { printf("- "); }
  | E '*' E  { printf("* "); }
  | E '/' E  { printf("/ "); }
  | '(' E ')'
  | ID       { printf("%c ", yylval); }
  ;

%%

void yyerror(char *s)
{
    printf("Invalid expression");
}

int main()
{
    printf("Enter infix expression: ");
    yyparse();
    return 0;

ast.y
%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct node {
    char v[20];
    struct node *l, *r;
} Node;

Node *root;

Node *create(char *v, Node *l, Node *r)
{
    Node *n = malloc(sizeof(Node));

    strcpy(n->v, v);
    n->l = l;
    n->r = r;

    return n;
}

int yylex(void);
void yyerror(const char *s);

void draw(Node *n, int d)
{
    int i;

    if (n == NULL)
        return;

    for (i = 0; i < d; i++)
        printf("  ");

    printf("%s\n", n->v);

    draw(n->l, d + 1);
    draw(n->r, d + 1);
}
%}

%union {
    Node *n;
    char *s;
}

%token <s> ID
%type <n> E T F

%%

E : E '+' T
    { $$ = root = create("+", $1, $3); }

  | E '-' T
    { $$ = root = create("-", $1, $3); }

  | T
    { $$ = $1; }
  ;

T : T '*' F
    { $$ = create("*", $1, $3); }

  | T '/' F
    { $$ = create("/", $1, $3); }

  | F
    { $$ = $1; }
  ;

F : '(' E ')'
    { $$ = $2; }

  | ID
    { $$ = create($1, NULL, NULL); }
  ;

%%

void yyerror(const char *s)
{
    printf("Invalid expression\n");
}

int main()
{
    printf("Enter expression: ");

    yyparse();

    draw(root, 0);

    return 0;
}
